/**
 * Executes a SQL query with parameters and stores the result for comparison.
 * Parameters are provided as comma-separated values that will replace placeholders in the query.
 * 
 * @param query the SQL query with ? placeholders for parameters
 * @param parameters comma-separated parameters to inject into the query
 * @param resultVariable name of the variable to store the query result
 */
@QAFTestStep(description = "I execute SQL query {query} with parameters {parameters} and store result in {resultVariable}")
public void executeSqlQueryWithParametersAndStore(String query, String parameters, String resultVariable) {
    logger.info("Executing SQL query with parameters and storing result in variable: " + resultVariable);
    
    try {
        // Parse the parameters
        String[] paramValues = parameters.split("\\s*,\\s*");
        List<Object> queryParams = new ArrayList<>();
        
        // Process each parameter
        for (String param : paramValues) {
            String trimmedParam = param.trim();
            
            // Check if the parameter is a reference to test data
            if (trimmedParam.startsWith("${") && trimmedParam.endsWith("}")) {
                String dataKey = trimmedParam.substring(2, trimmedParam.length() - 1);
                queryParams.add(TestContextManager.getTestDataValue(dataKey));
                logger.info("Parameter resolved from test data: " + dataKey + " = " + 
                           TestContextManager.getTestDataValue(dataKey));
            } else {
                // Direct value
                queryParams.add(trimmedParam);
                logger.info("Parameter direct value: " + trimmedParam);
            }
        }
        
        // Execute the query using the framework utility method
        List<Map<String, Object>> queryResults = SQLServerDBUtil.getResultsSet(query, queryParams);
        logger.info("Query executed successfully. Result size: " + queryResults.size());
        
        // Store the result in test context with the standard prefix for compatibility
        TestContextManager.setContext("db.query." + resultVariable, queryResults);
        Reporter.log("SQL query executed with parameters and result stored in variable: " + resultVariable);
        
    } catch (Exception e) {
        logger.error("Error executing SQL query with parameters: " + e.getMessage(), e);
        Reporter.log("Error executing SQL query with parameters: " + e.getMessage());
        throw new RuntimeException("Failed to execute SQL query with parameters: " + e.getMessage(), e);
    }
}

/**
 * Executes a SQL query and validates the result against a specific response field.

 * 
 * @param query SQL query to execute
 * @param parameters comma-separated query parameters
 * @param responsePath JSON path to the response field to validate
 * @param keyFields key fields for matching records

@QAFTestStep(description = "I validate response field {responsePath} against SQL query {query} with parameters {parameters} using key fields {keyFields}")
public void validateResponseFieldAgainstSqlQuery(String responsePath, String query, String parameters, String keyFields) {

    try {
        // Create a unique variable name for this query result
        String resultVariable = "sqlQueryResult_" + System.currentTimeMillis();

        executeSqlQueryWithParametersAndStore(query, parameters, resultVariable);

        Reporter.log("Error validating response against SQL query: " + e.getMessage());
        throw new RuntimeException("Failed to validate response against SQL query: " + e.getMessage(), e);
    }
}

/**
 * Enhances the existing validateResponseAgainstQueryResult method with support for nested JSON paths.
 * This method intercepts calls to validateResponseAgainstQueryResult and adds path extraction.
 * 
 * @param responseField the response field to validate (can be a nested path)
 * @param queryResult the query result variable name
 * @param keyFields comma-separated key fields for matching records
 */
@Override
public void validateResponseAgainstQueryResult(String responseField, String queryResult, String keyFields) {
    ApiResponse response = TestContextManager.getLastApiResponse();
    if (response == null) {
        Reporter.log("No API response available. Make sure to send a request first.", MessageTypes.Fail);
        Assert.fail("No API response available");
        return;
    }
    
    JSONObject jsonResponse = response.getJsonObject();
    
    // Check if this is a simple field or a nested path
    if (responseField.contains(".") || responseField.contains("[")) {
        // Handle nested path
        try {
            // Extract data from nested path
            Object responseData = extractFromJsonPath(jsonResponse, responseField);
            if (responseData == null) {
                Reporter.log("Response data not found at path: " + responseField, MessageTypes.Fail);
                Assert.fail("Response data not found at path: " + responseField);
                return;
            }
            
            // Create a temporary field in the JSON object to hold the extracted data
            String tempField = "__extracted_data_" + System.currentTimeMillis();
            
            // Add it to the response for regular processing
            if (responseData instanceof JSONArray) {
                jsonResponse.put(tempField, responseData);
            } else if (responseData instanceof JSONObject) {
                JSONArray array = new JSONArray();
                array.put(responseData);
                jsonResponse.put(tempField, array);
            } else {
                Reporter.log("Data at path '" + responseField + "' is not a valid JSON object or array", MessageTypes.Fail);
                Assert.fail("Invalid data type at path: " + responseField);
                return;
            }
            
            // Process using the existing method (which will use the stored temp field)
            super.validateResponseAgainstQueryResult(tempField, queryResult, keyFields);
            
            // Clean up
            jsonResponse.remove(tempField);
        } catch (Exception e) {
            logger.error("Error processing nested path: " + e.getMessage(), e);
            Reporter.log("Error processing nested path: " + e.getMessage(), MessageTypes.Fail);
            Assert.fail("Error processing nested path: " + e.getMessage());
        }
    } else {
        // Regular field, process normally
        super.validateResponseAgainstQueryResult(responseField, queryResult, keyFields);
    }
}

/**
 * Extracts data from a JSON object using a path expression.
 * Supports dot notation and array indexing (e.g., "empInfo[0].empDetails[*].deptId").
 * Special handling for [*] to get all array elements.
 */
private Object extractFromJsonPath(JSONObject json, String path) {
    if (path == null || path.isEmpty()) {
        return json;
    }
    
    // Simple case: direct field access
    if (!path.contains(".") && !path.contains("[")) {
        return json.has(path) ? json.get(path) : null;
    }
    
    String[] segments = path.split("\\.");
    Object current = json;
    
    for (String segment : segments) {
        if (current == null) {
            return null;
        }
        
        // Handle array access
        if (segment.contains("[") && segment.endsWith("]")) {
            String fieldName = segment.substring(0, segment.indexOf('['));
            String indexStr = segment.substring(segment.indexOf('[') + 1, segment.length() - 1);
            
            // Get the array field first
            if (current instanceof JSONObject) {
                JSONObject jsonObj = (JSONObject) current;
                if (!jsonObj.has(fieldName)) {
                    return null;
                }
                current = jsonObj.get(fieldName);
            } else {
                return null;
            }
            
            // Then handle array indexing
            if (current instanceof JSONArray) {
                JSONArray array = (JSONArray) current;
                
                // Handle wildcard [*] to get all array items
                if (indexStr.equals("*")) {
                    return array;
                }
                
                // Handle numeric index
                try {
                    int index = Integer.parseInt(indexStr);
                    if (index >= 0 && index < array.length()) {
                        current = array.get(index);
                    } else {
                        return null;
                    }
                } catch (NumberFormatException e) {
                    return null;
                }
            } else {
                return null;
            }
        } 
        // Simple field access
        else if (current instanceof JSONObject) {
            JSONObject jsonObj = (JSONObject) current;
            if (!jsonObj.has(segment)) {
                return null;
            }
            current = jsonObj.get(segment);
        } else {
            return null;
        }
    }
    
    return current;
}

/**
 * Compares field values between database query results and API response to identify missing or extra values.
 * Works with any field type (not just IDs) and handles complex nested response structures.
 * 
 * @param queryResult the variable name containing database query results
 * @param responsePath JSON path to the field values in the response (can be nested like "empInfo[*].empDetails[*].deptId")
 * @param dbField the field name in the database results
 */
@QAFTestStep(description = "I verify all {dbField} values from query result {queryResult} exist in response path {responsePath}")
public void verifyFieldValuesExistInResponse(String dbField, String queryResult, String responsePath) {
    logger.info("Verifying " + dbField + " values from database exist in response at " + responsePath);
    
    ApiResponse response = TestContextManager.getLastApiResponse();
    if (response == null) {
        Reporter.log("No API response available. Make sure to send a request first.", MessageTypes.Fail);
        Assert.fail("No API response available");
        return;
    }
    
    @SuppressWarnings("unchecked")
    List<Map<String, Object>> dbResults = (List<Map<String, Object>>) TestContextManager.getContext("db.query." + queryResult);
    if (dbResults == null || dbResults.isEmpty()) {
        Reporter.log("No database results available for key: " + queryResult, MessageTypes.Fail);
        Assert.fail("No database results available");
        return;
    }
    
    // Extract values from database results
    Set<String> dbValues = new HashSet<>();
    for (Map<String, Object> record : dbResults) {
        if (record.containsKey(dbField)) {
            Object valueObj = record.get(dbField);
            if (valueObj != null) {
                dbValues.add(valueObj.toString());
            }
        }
    }
    
    if (dbValues.isEmpty()) {
        Reporter.log("No " + dbField + " values found in database results", MessageTypes.Warn);
        return;
    }
    
    // Extract all values from the response at the specified path
    Set<String> responseValues = extractAllValuesFromPath(response.getJsonObject(), responsePath);
    
    // Find values in database but missing from response
    Set<String> missingValues = new HashSet<>(dbValues);
    missingValues.removeAll(responseValues);
    
    // Find values in response but not in database
    Set<String> extraValues = new HashSet<>(responseValues);
    extraValues.removeAll(dbValues);
    
    // Log results
    Reporter.log("Field comparison summary for '" + dbField + "':");
    Reporter.log("- Database values: " + dbValues.size());
    Reporter.log("- Response values: " + responseValues.size());
    
    if (missingValues.isEmpty()) {
        Reporter.log("✓ All database values exist in the response");
    } else {
        Reporter.log("✗ " + missingValues.size() + " database values are missing from the response:", 
                    MessageTypes.Fail);
        
        for (String value : missingValues) {
            Reporter.log("  - Missing: " + value);
        }
        
        Assert.fail("Database values missing from response");
    }
    
    if (!extraValues.isEmpty()) {
        Reporter.log("! Additional values found in response but not in database: " + extraValues.size());
        if (extraValues.size() <= 10) {
            for (String value : extraValues) {
                Reporter.log("  - Extra: " + value);
            }
        } else {
            Reporter.log("  (Too many additional values to display)");
        }
    }
}

/**
 * Helper method to extract all values from a JSON path, including handling nested arrays.
 * 
 * @param json the JSON object to extract from
 * @param path the path to extract (supports wildcards and nested arrays)
 * @return set of all string values found at the path
 */
private Set<String> extractAllValuesFromPath(JSONObject json, String path) {
    Set<String> results = new HashSet<>();
    
    // Handle wildcard paths like "empInfo[*].empDetails[*].deptId"
    if (path.contains("[*]")) {
        String[] parts = path.split("\\[\\*\\]\\.");
        
        // First part before the wildcard
        String rootPath = parts[0];
        // Get the array at the root path
        Object rootObj = extractFromJsonPath(json, rootPath);
        
        if (rootObj instanceof JSONArray) {
            JSONArray rootArray = (JSONArray) rootObj;
            
            // For each item in the root array
            for (int i = 0; i < rootArray.length(); i++) {
                // If we're at the last part, get the values
                if (parts.length == 1) {
                    Object value = rootArray.get(i);
                    if (value != null) {
                        results.add(value.toString());
                    }
                } 
                // Otherwise continue with the next part
                else {
                    Object item = rootArray.get(i);
                    if (item instanceof JSONObject) {
                        // Build the rest of the path
                        StringBuilder remainingPath = new StringBuilder();
                        for (int j = 1; j < parts.length; j++) {
                            remainingPath.append(parts[j]);
                            if (j < parts.length - 1) {
                                remainingPath.append("[*].");
                            }
                        }
                        
                        // Recursively extract values from the remaining path
                        results.addAll(extractAllValuesFromPath((JSONObject) item, remainingPath.toString()));
                    }
                }
            }
        }
    } 
    // Simple field access
    else {
        Object value = extractFromJsonPath(json, path);
        if (value != null) {
            if (value instanceof JSONArray) {
                JSONArray array = (JSONArray) value;
                for (int i = 0; i < array.length(); i++) {
                    results.add(array.get(i).toString());
                }
            } else {
                results.add(value.toString());
            }
        }
    }
    
    return results;
}

/**
 * Verifies that specific field values from database query results exist in the API response at a given path.
 * Supports custom mapping between database column names and response field names.
 * 
 * @param queryResult the variable name containing database query results
 * @param responsePath JSON path to the field values in the response (supports nested paths)
 * @param fieldMappings comma-separated mappings of DB columns to response fields (format: "dbField1:responseField1,dbField2:responseField2")
 */
@QAFTestStep(description = "I verify database fields from {queryResult} exist in response path {responsePath} with mappings {fieldMappings}")
public void verifyDbFieldsExistInResponsePath(String queryResult, String responsePath, String fieldMappings) {
    logger.info("Verifying database fields from " + queryResult + " exist in response at " + responsePath);
    
    ApiResponse response = TestContextManager.getLastApiResponse();
    if (response == null) {
        Reporter.log("No API response available. Make sure to send a request first.", MessageTypes.Fail);
        Assert.fail("No API response available");
        return;
    }
    
    @SuppressWarnings("unchecked")
    List<Map<String, Object>> dbResults = (List<Map<String, Object>>) TestContextManager.getContext("db.query." + queryResult);
    if (dbResults == null || dbResults.isEmpty()) {
        Reporter.log("No database results available for key: " + queryResult, MessageTypes.Fail);
        Assert.fail("No database results available");
        return;
    }
    
    // Parse field mappings
    Map<String, String> mappings = new HashMap<>();
    if (fieldMappings != null && !fieldMappings.isEmpty()) {
        String[] mappingPairs = fieldMappings.split("\\s*,\\s*");
        for (String pair : mappingPairs) {
            String[] parts = pair.split("\\s*:\\s*");
            if (parts.length == 2) {
                mappings.put(parts[0].trim(), parts[1].trim());
            }
        }
    }
    
    if (mappings.isEmpty()) {
        Reporter.log("No field mappings provided", MessageTypes.Fail);
        Assert.fail("No field mappings provided");
        return;
    }
    
    // For each DB field we want to check, collect all values including nulls
    Map<String, Set<String>> dbFieldValues = new HashMap<>();
    for (String dbField : mappings.keySet()) {
        Set<String> values = new HashSet<>();
        
        for (Map<String, Object> record : dbResults) {
            if (record.containsKey(dbField)) {
                Object valueObj = record.get(dbField);
                // Convert to string, preserving nulls as "null"
                String strValue = (valueObj == null) ? "null" : valueObj.toString();
                values.add(strValue);
            }
        }
        
        dbFieldValues.put(dbField, values);
    }
    
    // Extract response data at the specified path
    JSONObject jsonResponse = response.getJsonObject();
    Object responseData = extractFromJsonPath(jsonResponse, responsePath);
    
    if (responseData == null) {
        Reporter.log("Response data not found at path: " + responsePath, MessageTypes.Fail);
        Assert.fail("Response data not found at path: " + responsePath);
        return;
    }
    
    // Extract all values for each mapped response field
    Map<String, Set<String>> responseFieldValues = new HashMap<>();
    for (Map.Entry<String, String> mapping : mappings.entrySet()) {
        String responseField = mapping.getValue();
        responseFieldValues.put(responseField, new HashSet<>());
    }
    
    // Process response data and extract values
    if (responseData instanceof JSONArray) {
        JSONArray array = (JSONArray) responseData;
        for (int i = 0; i < array.length(); i++) {
            if (array.get(i) instanceof JSONObject) {
                JSONObject obj = array.getJSONObject(i);
                extractFieldValues(obj, "", responseFieldValues);
            }
        }
    } else if (responseData instanceof JSONObject) {
        JSONObject obj = (JSONObject) responseData;
        extractFieldValues(obj, "", responseFieldValues);
    }
    
    // Compare values for each field mapping
    boolean allMatch = true;
    for (Map.Entry<String, String> mapping : mappings.entrySet()) {
        String dbField = mapping.getKey();
        String responseField = mapping.getValue();
        
        Set<String> dbValues = dbFieldValues.get(dbField);
        Set<String> responseValues = responseFieldValues.get(responseField);
        
        // Find values in database but missing from response
        Set<String> missingValues = new HashSet<>(dbValues);
        missingValues.removeAll(responseValues);
        
        if (missingValues.isEmpty()) {
            Reporter.log("✓ All values for DB field '" + dbField + "' exist in response field '" + responseField + "'");
        } else {
            allMatch = false;
            Reporter.log("✗ " + missingValues.size() + " values for DB field '" + dbField + 
                        "' missing from response field '" + responseField + "':", MessageTypes.Fail);
            
            for (String value : missingValues) {
                Reporter.log("  - Missing: " + value);
            }
        }
    }
    
    if (!allMatch) {
        Assert.fail("Some database field values are missing from the response");
    }
}

/**
 * Extracts field values from a JSON object, including handling nested objects.
 * Supports dot notation in field names and preserves null values.
 * 
 * @param json the JSON object
 * @param prefix the current path prefix (for nested objects)
 * @param valueMap map of field names to value sets to populate
 */
private void extractFieldValues(JSONObject json, String prefix, Map<String, Set<String>> valueMap) {
    // Process direct fields
    for (String key : json.keySet()) {
        String fullPath = prefix.isEmpty() ? key : prefix + "." + key;
        
        // Check if this is a field we want
        if (valueMap.containsKey(fullPath)) {
            Object value = json.get(key);
            // Convert to string, preserving nulls as "null"
            String strValue = (value == JSONObject.NULL || value == null) ? "null" : value.toString();
            valueMap.get(fullPath).add(strValue);
        }
        
        // Handle nested objects
        Object value = json.get(key);
        if (value instanceof JSONObject) {
            extractFieldValues((JSONObject) value, fullPath, valueMap);
        } else if (value instanceof JSONArray) {
            JSONArray array = (JSONArray) value;
            for (int i = 0; i < array.length(); i++) {
                if (array.get(i) instanceof JSONObject) {
                    extractFieldValues(array.getJSONObject(i), fullPath, valueMap);
                }
            }
        }
    }
}
